# Pinned to the versions the test suite actually runs against — `backend/venv`,
# on Python 3.13. All 306 tests pass on this exact set.
#
# The previous pins claimed Django 4.2.13 while the venv ran 6.0.5, so a fresh
# `pip install -r requirements.txt` produced a stack that had never been tested:
# two major versions and four years of deprecations away from the code above it.
# Re-pin from `venv/Scripts/python -m pip freeze` whenever you upgrade, and
# re-run the suite before committing the change.
#
# Direct dependencies only — transitive ones (asgiref, sqlparse, tzdata, PyJWT,
# et_xmlfile, lxml, tablib, …) are left to the resolver so an upgrade here does
# not need a matching edit for each of them.

# --- SSO (Logto) ---
# Backs PyJWT's ES384 verification. Logto signs its ID tokens with ECDSA P-384 and
# PyJWT needs this to check the signature at all — without it `PyJWKClient` raises
# rather than verifying, so a Logto login would fail closed with a 500 instead of
# being rejected cleanly. PyJWT itself arrives transitively via simplejwt; it is
# pinned implicitly there and deliberately not repeated here.
cryptography==46.0.3

# because it is the intended replacement for that thread — until then, nothing
# imports it and no worker is expected to be running.
celery==5.6.3
